app = FastAPI(title="orders-api")

@app.exception_handler(RequestValidationError)
async def on_validation(req: Request, exc: RequestValidationError):
    # every 422 comes back in the same envelope as every 404/500
    return envelope(req, 422, "validation_error", exc.errors())

@app.get("/health")
async def health() -> dict:
    return {"status": "ok"}  # liveness probes hit this
